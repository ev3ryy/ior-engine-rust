struct EguiFieldVisitor<'a> {
    ui: &'a mut egui::Ui,
    changed: bool,
}

impl<'a> FieldVisitor for EguiFieldVisitor<'a> {
    fn visit_f32(&mut self, name: &'static str, val: &mut f32, meta: &ecs::FieldMeta) {
        let speed = meta.speed.unwrap_or(0.1);
        if let Some((min, max)) = meta.range {
            self.changed |= self.ui.add(egui::Slider::new(val, min..=max).text(name)).changed();
        } else {
            self.changed |= self.ui.add(egui::DragValue::new(val).speed(speed).prefix(format!("{}: ", name))).changed();
        }
    }

    fn visit_f64(&mut self, name: &'static str, val: &mut f64, meta: &ecs::FieldMeta) {
        let speed = meta.speed.unwrap_or(0.1) as f64;
        self.changed |= self.ui.add(egui::DragValue::new(val).speed(speed).prefix(format!("{}: ", name))).changed();
    }

    fn visit_i32(&mut self, name: &'static str, val: &mut i32, meta: &ecs::FieldMeta) {
        let speed = meta.speed.unwrap_or(1.0);
        if let Some((min, max)) = meta.range {
            self.changed |= self.ui.add(egui::Slider::new(val, (min as i32)..=(max as i32)).text(name)).changed();
        } else {
            self.changed |= self.ui.add(egui::DragValue::new(val).speed(speed).prefix(format!("{}: ", name))).changed();
        }
    }

    fn visit_u32(&mut self, name: &'static str, val: &mut u32, meta: &ecs::FieldMeta) {
        let speed = meta.speed.unwrap_or(1.0);
        self.changed |= self.ui.add(egui::DragValue::new(val).speed(speed).prefix(format!("{}: ", name))).changed();
    }

    fn visit_usize(&mut self, name: &'static str, val: &mut usize, meta: &ecs::FieldMeta) {
        let speed = meta.speed.unwrap_or(1.0);
        self.changed |= self.ui.add(egui::DragValue::new(val).speed(speed).prefix(format!("{}: ", name))).changed();
    }

    fn visit_bool(&mut self, name: &'static str, val: &mut bool, _meta: &ecs::FieldMeta) {
        self.changed |= self.ui.checkbox(val, name).changed();
    }

    fn visit_string(&mut self, name: &'static str, val: &mut String, _meta: &ecs::FieldMeta) {
        self.ui.horizontal(|ui| {
            ui.label(format!("{}:", name));
            self.changed |= ui.text_edit_singleline(val).changed();
        });
    }

    fn visit_vec2(&mut self, name: &'static str, val: &mut glam::Vec2, meta: &ecs::FieldMeta) {
        let speed = meta.speed.unwrap_or(0.1);
        self.ui.horizontal(|ui| {
            ui.label(format!("{}:", name));
            self.changed |= ui.add(egui::DragValue::new(&mut val.x).speed(speed).prefix("X: ")).changed();
            self.changed |= ui.add(egui::DragValue::new(&mut val.y).speed(speed).prefix("Y: ")).changed();
        });
    }

    fn visit_vec3(&mut self, name: &'static str, val: &mut glam::Vec3, meta: &ecs::FieldMeta) {
        let speed = meta.speed.unwrap_or(0.1);
        self.ui.horizontal(|ui| {
            ui.label(format!("{}:", name));
            self.changed |= ui.add(egui::DragValue::new(&mut val.x).speed(speed).prefix("X: ")).changed();
            self.changed |= ui.add(egui::DragValue::new(&mut val.y).speed(speed).prefix("Y: ")).changed();
            self.changed |= ui.add(egui::DragValue::new(&mut val.z).speed(speed).prefix("Z: ")).changed();
        });
    }

    fn visit_vec4(&mut self, name: &'static str, val: &mut glam::Vec4, meta: &ecs::FieldMeta) {
        let speed = meta.speed.unwrap_or(0.1);
        self.ui.horizontal(|ui| {
            ui.label(format!("{}:", name));
            self.changed |= ui.add(egui::DragValue::new(&mut val.x).speed(speed).prefix("X: ")).changed();
            self.changed |= ui.add(egui::DragValue::new(&mut val.y).speed(speed).prefix("Y: ")).changed();
            self.changed |= ui.add(egui::DragValue::new(&mut val.z).speed(speed).prefix("Z: ")).changed();
            self.changed |= ui.add(egui::DragValue::new(&mut val.w).speed(speed).prefix("W: ")).changed();
        });
    }

    fn visit_quat(&mut self, name: &'static str, val: &mut glam::Quat, _meta: &ecs::FieldMeta) {
        let (rx, ry, rz) = val.to_euler(glam::EulerRot::XYZ);
        let mut deg_x = rx.to_degrees();
        let mut deg_y = ry.to_degrees();
        let mut deg_z = rz.to_degrees();

        let mut rot_changed = false;
        self.ui.horizontal(|ui| {
            ui.label(format!("{}:", name));
            rot_changed |= ui.add(egui::DragValue::new(&mut deg_x).speed(0.5).prefix("R_X: ").suffix("°")).changed();
            rot_changed |= ui.add(egui::DragValue::new(&mut deg_y).speed(0.5).prefix("R_Y: ").suffix("°")).changed();
            rot_changed |= ui.add(egui::DragValue::new(&mut deg_z).speed(0.5).prefix("R_Z: ").suffix("°")).changed();
        });

        if rot_changed {
            *val = glam::Quat::from_euler(
                glam::EulerRot::XYZ,
                deg_x.to_radians(),
                deg_y.to_radians(),
                deg_z.to_radians(),
            );
            self.changed = true;
        }
    }

    fn visit_color3(&mut self, name: &'static str, val: &mut [f32; 3], _meta: &ecs::FieldMeta) {
        self.ui.horizontal(|ui| {
            ui.label(format!("{}:", name));
            self.changed |= ui.color_edit_button_rgb(val).changed();
        });
    }

    fn visit_f32_slice(&mut self, name: &'static str, val: &mut [f32; 3], meta: &ecs::FieldMeta) {
        let speed = meta.speed.unwrap_or(0.1);
        self.ui.horizontal(|ui| {
            ui.label(format!("{}:", name));
            self.changed |= ui.add(egui::DragValue::new(&mut val[0]).speed(speed).prefix("0: ")).changed();
            self.changed |= ui.add(egui::DragValue::new(&mut val[1]).speed(speed).prefix("1: ")).changed();
            self.changed |= ui.add(egui::DragValue::new(&mut val[2]).speed(speed).prefix("2: ")).changed();
        });
    }
}

pub fn pick_entity(
    ray: &Ray,
    world: &World,
    asset_server: &Arc<asset::AssetServer>,
) -> Option<u64> {
    let mut closest_hit: Option<(u64, f32)> = None;

    world.each_model_entity(|entity_id, global, model| {
        if let Some(asset_model) = asset_server.get::<render::model::Model>(model.id) {
            if let Some(mesh) = asset_model.meshes.get(model.mesh_index) {
                let aabb_min = Vec3::from(mesh.aabb_min);
                let aabb_max = Vec3::from(mesh.aabb_max);

                if let Some(t) = ray.intersect_obb(aabb_min, aabb_max, global.0) {
                    if t > 0.0 && closest_hit.map_or(true, |(_, min_t)| t < min_t) {
                        closest_hit = Some((entity_id, t));
                    }
                }
            }
        }
    });

    world.each_light_entity(|entity_id, global, _| {
        let center = global.translation();
        if let Some(t) = ray.intersect_sphere(center, 0.5) {
            if t > 0.0 && closest_hit.map_or(true, |(_, min_t)| t < min_t) {
                closest_hit = Some((entity_id, t));
            }
        }
    });

    world.each_camera_entity(|entity_id, global, _| {
        let center = global.translation();
        if let Some(t) = ray.intersect_sphere(center, 0.5) {
            if t > 0.0 && closest_hit.map_or(true, |(_, min_t)| t < min_t) {
                closest_hit = Some((entity_id, t));
            }
        }
    });

    closest_hit.map(|(id, _)| id)
}

fn project_3d_line(
    p0: Vec3,
    p1: Vec3,
    view_proj: Mat4,
    viewport_size: Vec2,
) -> Option<(Pos2, Pos2)> {
    let mut v0 = view_proj * p0.extend(1.0);
    let mut v1 = view_proj * p1.extend(1.0);

    let near = 0.001; // Порог ближней плоскости
    let in0 = v0.w > near;
    let in1 = v1.w > near;

    if !in0 && !in1 {
        return None;
    }

    // Отсекаем часть отрезка, оказавшуюся за камерой
    if !in0 {
        let t = (near - v0.w) / (v1.w - v0.w);
        v0 = v0.lerp(v1, t);
    } else if !in1 {
        let t = (near - v1.w) / (v0.w - v1.w);
        v1 = v1.lerp(v0, t);
    }

    let ndc0 = v0.truncate() / v0.w;
    let ndc1 = v1.truncate() / v1.w;

    let to_screen = |ndc: Vec3| -> Pos2 {
        let x = (ndc.x * 0.5 + 0.5) * viewport_size.x;
        let y = (1.0 - (ndc.y * 0.5 + 0.5)) * viewport_size.y;
        egui::pos2(x, y)
    };

    Some((to_screen(ndc0), to_screen(ndc1)))
}

// --- Генератор 3D-каркасов форм (Box, Sphere, Capsule, Cylinder) ---

pub fn draw_collider_wireframe(
    painter: &Painter,
    camera: &Camera,
    model_mat: Mat4,
    col: &ColliderComponent,
    viewport_size: Vec2,
) {
    let view = camera::rh::view::look_at_mat4(camera.eye, camera.target, camera.up);
    let proj = camera::rh::proj::directx::perspective(
        camera.fovy.to_radians(),
        camera.aspect,
        camera.znear,
        camera.zfar,
    );
    let view_proj = proj * view;

    // Зеленый для твердых тел, желтый для сенсоров/триггеров
    let color = if col.is_sensor {
        Color32::from_rgb(255, 200, 0)
    } else {
        Color32::from_rgb(0, 255, 120)
    };
    let stroke = Stroke::new(1.5, color);

    let mut lines: Vec<(Vec3, Vec3)> = Vec::new();

    match col.shape_type {
        ecs::ColliderShapeType::Cuboid => {
            let hx = col.size[0].max(0.001);
            let hy = col.size[1].max(0.001);
            let hz = col.size[2].max(0.001);

            let v = [
                Vec3::new(-hx, -hy, -hz), // 0
                Vec3::new(hx, -hy, -hz),  // 1
                Vec3::new(hx, hy, -hz),   // 2
                Vec3::new(-hx, hy, -hz),  // 3
                Vec3::new(-hx, -hy, hz),  // 4
                Vec3::new(hx, -hy, hz),   // 5
                Vec3::new(hx, hy, hz),    // 6
                Vec3::new(-hx, hy, hz),   // 7
            ];

            // 12 ребер параллелепипеда
            lines.extend_from_slice(&[
                (v[0], v[1]), (v[1], v[2]), (v[2], v[3]), (v[3], v[0]), // Задняя грань
                (v[4], v[5]), (v[5], v[6]), (v[6], v[7]), (v[7], v[4]), // Передняя грань
                (v[0], v[4]), (v[1], v[5]), (v[2], v[6]), (v[3], v[7]), // Соединения
            ]);
        }

        ecs::ColliderShapeType::Sphere => {
            let r = col.size[0].max(0.001);
            let segments = 24;

            for i in 0..segments {
                let theta0 = (i as f32 / segments as f32) * std::f32::consts::TAU;
                let theta1 = ((i + 1) as f32 / segments as f32) * std::f32::consts::TAU;

                let (s0, c0) = (theta0.sin() * r, theta0.cos() * r);
                let (s1, c1) = (theta1.sin() * r, theta1.cos() * r);

                // Окружность в плоскости XY
                lines.push((Vec3::new(c0, s0, 0.0), Vec3::new(c1, s1, 0.0)));
                // Окружность в плоскости XZ
                lines.push((Vec3::new(c0, 0.0, s0), Vec3::new(c1, 0.0, s1)));
                // Окружность в плоскости YZ
                lines.push((Vec3::new(0.0, c0, s0), Vec3::new(0.0, c1, s1)));
            }
        }

        ecs::ColliderShapeType::Capsule => {
            let r = col.size[0].max(0.001);
            let h = col.size[1].max(0.001);
            let segments = 20;

            for i in 0..segments {
                let theta0 = (i as f32 / segments as f32) * std::f32::consts::TAU;
                let theta1 = ((i + 1) as f32 / segments as f32) * std::f32::consts::TAU;

                let (s0, c0) = (theta0.sin() * r, theta0.cos() * r);
                let (s1, c1) = (theta1.sin() * r, theta1.cos() * r);

                // Верхнее и нижнее кольца
                lines.push((Vec3::new(c0, h, s0), Vec3::new(c1, h, s1)));
                lines.push((Vec3::new(c0, -h, s0), Vec3::new(c1, -h, s1)));

                // Полусферы (арки)
                let phi0 = (i as f32 / segments as f32) * std::f32::consts::PI;
                let phi1 = ((i + 1) as f32 / segments as f32) * std::f32::consts::PI;
                let (sp0, cp0) = (phi0.sin() * r, phi0.cos() * r);
                let (sp1, cp1) = (phi1.sin() * r, phi1.cos() * r);

                lines.push((Vec3::new(sp0, h + cp0, 0.0), Vec3::new(sp1, h + cp1, 0.0)));
                lines.push((Vec3::new(0.0, h + cp0, sp0), Vec3::new(0.0, h + cp1, sp1)));
                lines.push((Vec3::new(sp0, -h - cp0, 0.0), Vec3::new(sp1, -h - cp1, 0.0)));
                lines.push((Vec3::new(0.0, -h - cp0, sp0), Vec3::new(0.0, -h - cp1, sp1)));
            }

            // 4 вертикальные линии
            lines.push((Vec3::new(r, h, 0.0), Vec3::new(r, -h, 0.0)));
            lines.push((Vec3::new(-r, h, 0.0), Vec3::new(-r, -h, 0.0)));
            lines.push((Vec3::new(0.0, h, r), Vec3::new(0.0, -h, r)));
            lines.push((Vec3::new(0.0, h, -r), Vec3::new(0.0, -h, -r)));
        }

        ecs::ColliderShapeType::Cylinder => {
            let r = col.size[0].max(0.001);
            let h = col.size[1].max(0.001);
            let segments = 24;

            for i in 0..segments {
                let theta0 = (i as f32 / segments as f32) * std::f32::consts::TAU;
                let theta1 = ((i + 1) as f32 / segments as f32) * std::f32::consts::TAU;

                let (s0, c0) = (theta0.sin() * r, theta0.cos() * r);
                let (s1, c1) = (theta1.sin() * r, theta1.cos() * r);

                // Верхнее и нижнее основания
                lines.push((Vec3::new(c0, h, s0), Vec3::new(c1, h, s1)));
                lines.push((Vec3::new(c0, -h, s0), Vec3::new(c1, -h, s1)));
            }

            // 4 вертикальные образующие
            lines.push((Vec3::new(r, h, 0.0), Vec3::new(r, -h, 0.0)));
            lines.push((Vec3::new(-r, h, 0.0), Vec3::new(-r, -h, 0.0)));
            lines.push((Vec3::new(0.0, h, r), Vec3::new(0.0, -h, r)));
            lines.push((Vec3::new(0.0, h, -r), Vec3::new(0.0, -h, -r)));
        }
    }

    // Трансформируем линии в мировые координаты и проецируем на 2D-экран
    for (l0, l1) in lines {
        let p0 = model_mat.transform_point3(l0);
        let p1 = model_mat.transform_point3(l1);

        if let Some((s0, s1)) = project_3d_line(p0, p1, view_proj, viewport_size) {
            painter.line_segment([s0, s1], stroke);
        }
    }
}

fn draw_entity_node(
    ui: &mut Ui,
    world: &mut World,
    entity_id: u64,
    selected_entity: &mut Option<u64>,
) {
    if !world.is_alive(entity_id) {
        return;
    }

    let is_selected = *selected_entity == Some(entity_id);
    let name = world.get_entity_name(entity_id);
    let display_name = if name.is_empty() {
        format!("Entity {}", entity_id)
    } else {
        name
    };

    let has_light = world.has_light(entity_id);
    let has_model = world.has_model(entity_id);
    let has_camera = world.has_camera(entity_id);

    let icon = if has_light {
        "💡"
    } else if has_model {
        "🔷"
    } else if has_camera {
        "📷"
    } else {
        "📁"
    };

    let header_text = format!("{} {} ({})", icon, display_name, entity_id);
    let header_id = ui.make_persistent_id(entity_id);
    let has_children = world.has_children(entity_id);

    if has_children {
        egui::collapsing_header::CollapsingState::load_with_default_open(ui.ctx(), header_id, true)
            .show_header(ui, |ui| {
                if ui.selectable_label(is_selected, &header_text).clicked() {
                    if is_selected {
                        *selected_entity = None;
                    } else {
                        *selected_entity = Some(entity_id);
                    }
                }
            })
            .body(|ui| {
                let children = world.get_children(entity_id);
                for child_id in children {
                    draw_entity_node(ui, world, child_id, selected_entity);
                }
            });
    } else {
        if ui.selectable_label(is_selected, &header_text).clicked() {
            if is_selected {
                *selected_entity = None;
            } else {
                *selected_entity = Some(entity_id);
            }
        }
    }
}

pub fn draw_postprocess_window(ctx: &Context, pp: &mut PostProcessSettings) {
    egui::Window::new("Post-Processing")
        .default_pos([870.0, 10.0])
        .default_size([280.0, 400.0])
        .show(ctx, |ui| {
            ui.horizontal(|ui| {
                ui.checkbox(&mut pp.enabled, "Enable Post-Processing");
                ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                    if ui.button("↺ Reset").on_hover_text("Reset to Defaults").clicked() {
                        *pp = PostProcessSettings::default();
                    }
                });
            });
            ui.separator();

            if !pp.enabled {
                ui.label(egui::RichText::new("Post-processing pipeline is bypassed.").italics());
                return;
            }

            egui::ScrollArea::vertical().show(ui, |ui| {
                ui.group(|ui| {
                    ui.label(egui::RichText::new("📷 Tonemapping & Exposure").strong());
                    egui::ComboBox::from_label("Tonemap")
                        .selected_text(match pp.tonemapping_mode {
                            TonemappingMode::None => "None (Clamp)",
                            TonemappingMode::ACES => "ACES Filmic",
                            TonemappingMode::Reinhard => "Reinhard",
                        })
                        .show_ui(ui, |ui| {
                            ui.selectable_value(&mut pp.tonemapping_mode, TonemappingMode::ACES, "ACES Filmic");
                            ui.selectable_value(&mut pp.tonemapping_mode, TonemappingMode::Reinhard, "Reinhard");
                            ui.selectable_value(&mut pp.tonemapping_mode, TonemappingMode::None, "None (Clamp)");
                        });

                    ui.add(egui::Slider::new(&mut pp.exposure, 0.01..=5.0).text("Exposure").logarithmic(true));
                    ui.checkbox(&mut pp.manual_gamma_correction, "Manual Gamma (1 / 2.2)");
                });
                ui.add_space(5.0);

                ui.group(|ui| {
                    ui.horizontal(|ui| {
                        ui.label(egui::RichText::new("✨ Bloom (Dual Filtering)").strong());
                        ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                            ui.checkbox(&mut pp.bloom_enabled, "");
                        });
                    });

                    if pp.bloom_enabled {
                        ui.add(egui::Slider::new(&mut pp.bloom_intensity, 0.0..=0.5).text("Intensity"));
                        ui.add(egui::Slider::new(&mut pp.bloom_threshold, 0.0..=10.0).text("Threshold"));
                        ui.add(egui::Slider::new(&mut pp.bloom_knee, 0.0..=1.0).text("Knee (Softness)"));
                        ui.add(egui::Slider::new(&mut pp.bloom_scatter, 0.1..=2.0).text("Scatter / Radius"));
                    }
                });
                ui.add_space(5.0);

                ui.group(|ui| {
                    ui.label(egui::RichText::new("🎨 Color Grading").strong());
                    ui.add(egui::Slider::new(&mut pp.contrast, 0.5..=2.0).text("Contrast"));
                    ui.add(egui::Slider::new(&mut pp.saturation, 0.0..=2.0).text("Saturation"));
                });
                ui.add_space(5.0);

                ui.group(|ui| {
                    ui.horizontal(|ui| {
                        ui.label(egui::RichText::new("⭕ Vignette").strong());
                        ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                            ui.checkbox(&mut pp.vignette_enabled, "");
                        });
                    });

                    if pp.vignette_enabled {
                        ui.add(egui::Slider::new(&mut pp.vignette_intensity, 0.0..=1.0).text("Intensity"));
                        ui.add(egui::Slider::new(&mut pp.vignette_radius, 0.1..=1.5).text("Radius"));
                        ui.add(egui::Slider::new(&mut pp.vignette_smoothness, 0.01..=1.0).text("Smoothness"));
                    }
                });
                ui.add_space(5.0);

                ui.group(|ui| {
                    ui.horizontal(|ui| {
                        ui.label(egui::RichText::new("🎞 Film Grain").strong());
                        ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                            ui.checkbox(&mut pp.grain_enabled, "");
                        });
                    });

                    if pp.grain_enabled {
                        ui.add(egui::Slider::new(&mut pp.grain_intensity, 0.0..=0.1).text("Intensity"));
                    }
                });
            });
        });
}

pub fn draw_editor_ui(
    ctx: &Context,
    world: &mut World,
    camera: &mut Camera,
    postprocess: &mut PostProcessSettings,
    metrics: &EngineMetrics,
    load_requests: &mut Vec<String>,
    selected_entity: &mut Option<u64>,
    gizmo: &mut Gizmo,
    gizmo_mode: &mut EditorGizmoMode,
    gizmo_orientation: &mut GizmoOrientation,
    play_state: PlayState,
    play_action: &mut PlayControlAction,
    asset_server: &Arc<asset::AssetServer>,
) {
    let mut clear_requested = false;

    draw_postprocess_window(ctx, postprocess);

    let is_editing = play_state != PlayState::Playing;
    let mut gizmo_was_dragged = false;

    egui::Area::new(egui::Id::new("viewport_canvas"))
        .order(egui::Order::Background)
        .fixed_pos([0.0, 0.0])
        .show(ctx, |ui| {
            let screen_rect = ui
                .ctx()
                .input(|i| i.raw.screen_rect)
                .unwrap_or_else(|| egui::Rect::from_min_size(egui::Pos2::ZERO, egui::vec2(1280.0, 720.0)));

            let response = ui.allocate_rect(screen_rect, egui::Sense::click());
            let painter = ui.painter();
            let viewport_size = glam::Vec2::new(screen_rect.width(), screen_rect.height());

            // --- Отрисовка каркаса коллайдера выделенного объекта ---
            if is_editing {
                if let Some(entity_id) = *selected_entity {
                    if let Some(model_mat) = world.get_global_transform(entity_id) {
                        if world.has_collider(entity_id) {
                            world.get_collider_mut(entity_id, |col| {
                                draw_collider_wireframe(
                                    painter,
                                    camera,
                                    model_mat,
                                    col,
                                    viewport_size,
                                );
                            });
                        }
                    }
                }
            }

            // --- 3D Transform Gizmo ---
            if is_editing {
                if let Some(entity_id) = *selected_entity {
                    if let Some(model_mat) = world.get_global_transform(entity_id) {
                        let eye = DVec3::new(camera.eye.x as f64, camera.eye.y as f64, camera.eye.z as f64);
                        let target = DVec3::new(camera.target.x as f64, camera.target.y as f64, camera.target.z as f64);
                        let up = DVec3::new(camera.up.x as f64, camera.up.y as f64, camera.up.z as f64);

                        let view_mat = DMat4::look_at_rh(eye, target, up);
                        let proj_mat = DMat4::perspective_infinite_rh(
                            (camera.fovy as f64).to_radians(),
                            camera.aspect as f64,
                            camera.znear as f64,
                        );

                        let modes: EnumSet<GizmoMode> = match gizmo_mode {
                            EditorGizmoMode::Translate => GizmoMode::all().into_iter().filter(|m| m.is_translate()).collect(),
                            EditorGizmoMode::Rotate => GizmoMode::all().into_iter().filter(|m| m.is_rotate()).collect(),
                            EditorGizmoMode::Scale => GizmoMode::all().into_iter().filter(|m| m.is_scale()).collect(),
                            EditorGizmoMode::All => GizmoMode::all(),
                        };

                        gizmo.update_config(GizmoConfig {
                            view_matrix: view_mat.into(),
                            projection_matrix: proj_mat.into(),
                            modes,
                            orientation: *gizmo_orientation,
                            ..Default::default()
                        });

                        let (scale, rot, pos) = model_mat.to_scale_rotation_translation();
                        let transform = GizmoTransform::from_scale_rotation_translation(
                            DVec3::new(scale.x as f64, scale.y as f64, scale.z as f64),
                            DQuat::from_xyzw(rot.x as f64, rot.y as f64, rot.z as f64, rot.w as f64),
                            DVec3::new(pos.x as f64, pos.y as f64, pos.z as f64),
                        );

                        if let Some((_result, new_transforms)) = gizmo.interact(ui, &[transform]) {
                            gizmo_was_dragged = true;
                            if let Some(new_t) = new_transforms.first() {
                                let s = new_t.scale;
                                let r = new_t.rotation;
                                let p = new_t.translation;

                                let new_global = glam::Mat4::from_scale_rotation_translation(
                                    glam::Vec3::new(s.x as f32, s.y as f32, s.z as f32),
                                    glam::Quat::from_xyzw(r.v.x as f32, r.v.y as f32, r.v.z as f32, r.s as f32),
                                    glam::Vec3::new(p.x as f32, p.y as f32, p.z as f32),
                                );

                                world.set_transform_from_global_matrix(entity_id, new_global);
                                world.update_transforms();
                            }
                        }
                    }
                }
            }

            // --- Выбор объекта кликом ЛКМ ---
            if is_editing && response.clicked() && !gizmo_was_dragged {
                if let Some(mouse_pos) = response.interact_pointer_pos() {
                    let click_pos = glam::Vec2::new(mouse_pos.x - screen_rect.min.x, mouse_pos.y - screen_rect.min.y);
                    let ray = camera.screen_to_ray(click_pos, viewport_size);

                    if let Some(hit_entity) = pick_entity(&ray, world, asset_server) {
                        *selected_entity = Some(hit_entity);
                    } else {
                        *selected_entity = None;
                    }
                }
            }
        });

    // Toolbar
    egui::Window::new("Toolbar")
        .default_pos([10.0, 10.0])
        .title_bar(false)
        .resizable(false)
        .show(ctx, |ui| {
            ui.horizontal(|ui| {
                let is_running = matches!(play_state, PlayState::Playing | PlayState::Simulating);
                let is_paused = matches!(play_state, PlayState::Paused { .. });

                if is_running || is_paused {
                    if ui.button(egui::RichText::new("⏹ Stop").color(egui::Color32::LIGHT_RED)).clicked() {
                        *play_action = PlayControlAction::Stop;
                    }
                    if ui.button(if is_paused { "▶ Resume" } else { "⏸ Pause" }).clicked() {
                        *play_action = if is_paused { PlayControlAction::Resume } else { PlayControlAction::Pause };
                    }
                    if is_paused && ui.button("⏭ Step").clicked() {
                        *play_action = PlayControlAction::Step;
                    }

                    ui.separator();
                    if play_state == PlayState::Simulating {
                        ui.label(egui::RichText::new("⚡ SIMULATING").color(egui::Color32::from_rgb(100, 200, 255)).strong());
                    } else if play_state == PlayState::Playing {
                        ui.label(egui::RichText::new("🎮 PLAYING").color(egui::Color32::from_rgb(100, 255, 100)).strong());
                    }
                } else {
                    if ui.button(egui::RichText::new("▶ Play").color(egui::Color32::GREEN)).on_hover_text("Play from Game Camera with Game Controls").clicked() {
                        *play_action = PlayControlAction::Start;
                    }
                    if ui.button(egui::RichText::new("⚡ Simulate").color(egui::Color32::from_rgb(100, 200, 255))).on_hover_text("Run scripts & physics while retaining Editor Camera & Tools").clicked() {
                        *play_action = PlayControlAction::Simulate;
                    }
                }

                ui.separator();

                ui.menu_button("File", |ui| {
                    if ui.button("Save Scene (F5)").clicked() {
                        if let Err(e) = serialization::save_world(world, "res/saved_scene.ron") {
                            log::error!("Failed to save scene: {e}");
                        } else {
                            log::info!("Scene saved to res/saved_scene.ron");
                        }
                    }
                    if ui.button("Load Scene (F9)").clicked() {
                        if let Err(e) = serialization::load_world(world, "res/saved_scene.ron") {
                            log::error!("Failed to load scene: {e}");
                        } else {
                            *selected_entity = None;
                            log::info!("Scene loaded from res/saved_scene.ron");
                        }
                    }
                    if ui.button("Clear Scene").clicked() {
                        clear_requested = true;
                    }
                    ui.separator();
                    if ui.button("Quit").clicked() {
                        std::process::exit(0);
                    }
                });

                ui.menu_button("➕ Spawn", |ui| {
                    if ui.button("Empty Entity").clicked() {
                        let id = world.spawn_empty("Empty Entity");
                        *selected_entity = Some(id);
                    }
                    if ui.button("📷 Camera").clicked() {
                        let id = world.spawn_empty("Camera");
                        let pos = camera.eye;
                        world.add_transform(id, Transform::from_translation(pos));
                        world.add_camera(id, CameraComponent::default());
                        *selected_entity = Some(id);
                    }
                    if ui.button("💡 Point Light").clicked() {
                        let id = world.spawn_empty("Point Light");
                        let pos = camera.eye + (camera.target - camera.eye).normalize() * 3.0;
                        world.add_transform(id, Transform::from_translation(pos));
                        world.add_light(id, LightComponent {
                            light_type: LightType::Point,
                            color: [1.0, 1.0, 1.0],
                            intensity: 10.0,
                            range: 15.0,
                            inner_angle: 0.0,
                            outer_angle: 0.0,
                        });
                        *selected_entity = Some(id);
                    }
                    if ui.button("☀️ Directional Light").clicked() {
                        let id = world.spawn_empty("Directional Light");
                        let pos = glam::Vec3::new(0.0, 10.0, 0.0);
                        let rot = glam::Quat::from_rotation_x(-0.7);
                        world.add_transform(id, Transform {
                            position: pos,
                            rotation: rot,
                            scale: glam::Vec3::ONE,
                        });
                        world.add_light(id, LightComponent {
                            light_type: LightType::Directional,
                            color: [1.0, 0.95, 0.8],
                            intensity: 5.0,
                            range: 100.0,
                            inner_angle: 0.0,
                            outer_angle: 0.0,
                        });
                        *selected_entity = Some(id);
                    }
                });

                ui.separator();

                ui.selectable_value(gizmo_mode, EditorGizmoMode::Translate, "Translate (W)");
                ui.selectable_value(gizmo_mode, EditorGizmoMode::Rotate, "Rotate (E)");
                ui.selectable_value(gizmo_mode, EditorGizmoMode::Scale, "Scale (R)");
                ui.selectable_value(gizmo_mode, EditorGizmoMode::All, "All (T)");

                ui.separator();

                if ui.button(match gizmo_orientation {
                    GizmoOrientation::Global => "🌐 Global (Q)",
                    GizmoOrientation::Local => "📦 Local (Q)",
                }).clicked() {
                    *gizmo_orientation = match gizmo_orientation {
                        GizmoOrientation::Global => GizmoOrientation::Local,
                        GizmoOrientation::Local => GizmoOrientation::Global,
                    };
                }

                ui.separator();

                ui.menu_button("View", |ui| {
                    ui.checkbox(&mut true, "Show Inspector");
                    ui.checkbox(&mut true, "Show Metrics");
                });
            });
        });

    // Scene Hierarchy Window
    egui::Window::new("Scene Hierarchy")
        .default_pos([10.0, 50.0])
        .default_size([250.0, 450.0])
        .show(ctx, |ui| {
            ui.horizontal(|ui| {
                if ui.button("➕ Empty").clicked() {
                    let id = world.spawn_empty("New Entity");
                    *selected_entity = Some(id);
                }
                if let Some(parent_id) = *selected_entity {
                    if ui.button("➕ Child").clicked() {
                        let id = world.spawn_child(parent_id, "Child Entity");
                        *selected_entity = Some(id);
                    }
                }
            });
            ui.separator();

            egui::ScrollArea::vertical().show(ui, |ui| {
                let roots = world.get_root_entities();
                if roots.is_empty() {
                    ui.label(egui::RichText::new("Scene is empty.").italics());
                } else {
                    for root_id in roots {
                        draw_entity_node(ui, world, root_id, selected_entity);
                    }
                }
            });
        });

    // Inspector Window
    egui::Window::new("Inspector")
        .default_pos([270.0, 50.0])
        .default_size([300.0, 500.0])
        .show(ctx, |ui| {
            if let Some(entity_id) = *selected_entity {
                if !world.is_alive(entity_id) {
                    *selected_entity = None;
                    return;
                }

                ui.horizontal(|ui| {
                    let mut name = world.get_entity_name(entity_id);
                    if ui.text_edit_singleline(&mut name).changed() {
                        world.set_entity_name(entity_id, &name);
                    }
                    if ui.button("✖ Deselect").clicked() {
                        *selected_entity = None;
                        return;
                    }
                });
                ui.label(format!("Entity ID: {}", entity_id));
                ui.separator();

                egui::ScrollArea::vertical().show(ui, |ui| {
                    let has_transform = world.has_transform(entity_id);
                    let has_camera = world.has_camera(entity_id);
                    let has_light = world.has_light(entity_id);
                    let has_model = world.has_model(entity_id);
                    let has_rigid_body = world.has_rigid_body(entity_id);
                    let has_collider = world.has_collider(entity_id);

                    // transform
                    if has_transform {
                        ui.group(|ui| {
                            ui.horizontal(|ui| {
                                ui.label(egui::RichText::new("📍 Transform").strong());
                                ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                                    if ui.button("❌").on_hover_text("Remove Transform").clicked() {
                                        world.remove_transform(entity_id);
                                    }
                                });
                            });

                            let mut transform_changed = false;

                            world.get_transform_mut(entity_id, |transform| {
                                ui.horizontal(|ui| {
                                    transform_changed |= ui.add(egui::DragValue::new(&mut transform.position.x).speed(0.1).prefix("X: ")).changed();
                                    transform_changed |= ui.add(egui::DragValue::new(&mut transform.position.y).speed(0.1).prefix("Y: ")).changed();
                                    transform_changed |= ui.add(egui::DragValue::new(&mut transform.position.z).speed(0.1).prefix("Z: ")).changed();
                                });

                                let (rx, ry, rz) = transform.rotation.to_euler(glam::EulerRot::XYZ);
                                let mut deg_x = rx.to_degrees();
                                let mut deg_y = ry.to_degrees();
                                let mut deg_z = rz.to_degrees();

                                let mut rot_changed = false;
                                ui.horizontal(|ui| {
                                    rot_changed |= ui.add(egui::DragValue::new(&mut deg_x).speed(0.5).prefix("R_X: ").suffix("°")).changed();
                                    rot_changed |= ui.add(egui::DragValue::new(&mut deg_y).speed(0.5).prefix("R_Y: ").suffix("°")).changed();
                                    rot_changed |= ui.add(egui::DragValue::new(&mut deg_z).speed(0.5).prefix("R_Z: ").suffix("°")).changed();
                                });

                                if rot_changed {
                                    transform.rotation = glam::Quat::from_euler(
                                        glam::EulerRot::XYZ,
                                        deg_x.to_radians(),
                                        deg_y.to_radians(),
                                        deg_z.to_radians(),
                                    );
                                    transform_changed = true;
                                }

                                ui.horizontal(|ui| {
                                    transform_changed |= ui.add(egui::DragValue::new(&mut transform.scale.x).speed(0.1).prefix("S_X: ")).changed();
                                    transform_changed |= ui.add(egui::DragValue::new(&mut transform.scale.y).speed(0.1).prefix("S_Y: ")).changed();
                                    transform_changed |= ui.add(egui::DragValue::new(&mut transform.scale.z).speed(0.1).prefix("S_Z: ")).changed();
                                });
                            });

                            if transform_changed {
                                world.update_transforms();
                            }
                        });
                        ui.add_space(5.0);
                    }

                    // camera
                    if has_camera {
                        ui.group(|ui| {
                            ui.horizontal(|ui| {
                                ui.label(egui::RichText::new("📷 Camera Component").strong());
                                ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                                    if ui.button("❌").on_hover_text("Remove Camera").clicked() {
                                        world.remove_camera(entity_id);
                                    }
                                });
                            });

                            world.get_camera_mut(entity_id, |camera| {
                                ui.checkbox(&mut camera.is_active, "Active");
                                ui.add(egui::Slider::new(&mut camera.fovy, 10.0..=120.0).text("FOV"));
                                ui.add(egui::Slider::new(&mut camera.znear, 0.01..=10.0).text("Z Near"));
                                ui.add(egui::Slider::new(&mut camera.zfar, 10.0..=10000.0).text("Z Far"));
                                ui.add(egui::DragValue::new(&mut camera.priority).prefix("Priority: "));
                            });
                        });
                        ui.add_space(5.0);
                    }

                    // light
                    if has_light {
                        ui.group(|ui| {
                            ui.horizontal(|ui| {
                                ui.label(egui::RichText::new("💡 Light Component").strong());
                                ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                                    if ui.button("❌").on_hover_text("Remove Light").clicked() {
                                        world.remove_light(entity_id);
                                    }
                                });
                            });

                            world.get_light_mut(entity_id, |light| {
                                egui::ComboBox::from_label("Type")
                                    .selected_text(match light.light_type {
                                        LightType::Directional => "Directional",
                                        LightType::Point => "Point",
                                        LightType::Spot => "Spot",
                                    })
                                    .show_ui(ui, |ui| {
                                        ui.selectable_value(&mut light.light_type, LightType::Point, "Point");
                                        ui.selectable_value(&mut light.light_type, LightType::Directional, "Directional");
                                        ui.selectable_value(&mut light.light_type, LightType::Spot, "Spot");
                                    });

                                ui.horizontal(|ui| {
                                    ui.label("Color:");
                                    let mut color = [light.color[0], light.color[1], light.color[2]];
                                    ui.color_edit_button_rgb(&mut color);
                                    light.color = color;
                                });

                                ui.add(egui::Slider::new(&mut light.intensity, 0.0..=1000.0).text("Intensity"));

                                match light.light_type {
                                    LightType::Point | LightType::Spot => {
                                        ui.add(egui::Slider::new(&mut light.range, 0.1..=200.0).text("Range"));
                                    }
                                    _ => {}
                                }

                                if let LightType::Spot = light.light_type {
                                    let mut inner_deg = light.inner_angle.to_degrees();
                                    let mut outer_deg = light.outer_angle.to_degrees();

                                    ui.add(egui::Slider::new(&mut inner_deg, 0.0..=90.0).text("Inner Cone"));
                                    ui.add(egui::Slider::new(&mut outer_deg, 0.0..=90.0).text("Outer Cone"));

                                    light.inner_angle = inner_deg.to_radians();
                                    light.outer_angle = outer_deg.to_radians().max(light.inner_angle + 0.01);
                                }
                            });
                        });
                        ui.add_space(5.0);
                    }

                    // mesh
                    if has_model {
                        ui.group(|ui| {
                            ui.horizontal(|ui| {
                                ui.label(egui::RichText::new("🔷 Mesh Component").strong());
                                ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                                    if ui.button("❌").on_hover_text("Remove Mesh").clicked() {
                                        world.remove_model(entity_id);
                                    }
                                });
                            });

                            world.get_model(entity_id, |model| {
                                ui.label(format!("Model ID: {:?}", model.id.path()));
                                ui.label(format!("Mesh Index: {}", model.mesh_index));
                            });
                        });
                        ui.add_space(5.0);
                    }

                    // rigidbody
                    if has_rigid_body {
                        ui.group(|ui| {
                            ui.horizontal(|ui| {
                                ui.label(egui::RichText::new("📦 RigidBody").strong());
                                ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                                    if ui.button("❌").on_hover_text("Remove RigidBody").clicked() {
                                        world.remove_rigid_body(entity_id);
                                    }
                                });
                            });

                            world.get_rigid_body_mut(entity_id, |rb| {
                                egui::ComboBox::from_label("Type")
                                    .selected_text(match rb.body_type {
                                        ecs::BodyType::Dynamic => "Dynamic",
                                        ecs::BodyType::Fixed => "Fixed (Static)",
                                        ecs::BodyType::KinematicPositionBased => "Kinematic (Position)",
                                        ecs::BodyType::KinematicVelocityBased => "Kinematic (Velocity)",
                                    })
                                    .show_ui(ui, |ui| {
                                        ui.selectable_value(&mut rb.body_type, ecs::BodyType::Dynamic, "Dynamic");
                                        ui.selectable_value(&mut rb.body_type, ecs::BodyType::Fixed, "Fixed (Static)");
                                        ui.selectable_value(&mut rb.body_type, ecs::BodyType::KinematicPositionBased, "Kinematic (Position)");
                                        ui.selectable_value(&mut rb.body_type, ecs::BodyType::KinematicVelocityBased, "Kinematic (Velocity)");
                                    });

                                if rb.body_type == ecs::BodyType::Dynamic {
                                    ui.add(egui::Slider::new(&mut rb.mass, 0.01..=1000.0).text("Mass (kg)"));
                                    ui.add(egui::Slider::new(&mut rb.gravity_scale, -10.0..=10.0).text("Gravity Scale"));
                                    ui.add(egui::Slider::new(&mut rb.linear_damping, 0.0..=10.0).text("Linear Damping"));
                                    ui.add(egui::Slider::new(&mut rb.angular_damping, 0.0..=10.0).text("Angular Damping"));
                                    ui.checkbox(&mut rb.ccd_enabled, "CCD (Continuous Collision)");
                                    ui.checkbox(&mut rb.lock_rotations, "Lock Rotations");
                                }
                            });
                        });
                        ui.add_space(5.0);
                    }

                    // collider
                    if has_collider {
                        ui.group(|ui| {
                            ui.horizontal(|ui| {
                                ui.label(egui::RichText::new("🛡 Collider").strong());
                                ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                                    if ui.button("❌").on_hover_text("Remove Collider").clicked() {
                                        world.remove_collider(entity_id);
                                    }
                                });
                            });

                            world.get_collider_mut(entity_id, |col| {
                                egui::ComboBox::from_label("Shape")
                                    .selected_text(match col.shape_type {
                                        ecs::ColliderShapeType::Cuboid => "Cuboid (Box)",
                                        ecs::ColliderShapeType::Sphere => "Sphere",
                                        ecs::ColliderShapeType::Capsule => "Capsule",
                                        ecs::ColliderShapeType::Cylinder => "Cylinder",
                                    })
                                    .show_ui(ui, |ui| {
                                        ui.selectable_value(&mut col.shape_type, ecs::ColliderShapeType::Cuboid, "Cuboid (Box)");
                                        ui.selectable_value(&mut col.shape_type, ecs::ColliderShapeType::Sphere, "Sphere");
                                        ui.selectable_value(&mut col.shape_type, ecs::ColliderShapeType::Capsule, "Capsule");
                                        ui.selectable_value(&mut col.shape_type, ecs::ColliderShapeType::Cylinder, "Cylinder");
                                    });

                                match col.shape_type {
                                    ecs::ColliderShapeType::Cuboid => {
                                        ui.horizontal(|ui| {
                                            ui.label("Half Extents:");
                                            ui.add(egui::DragValue::new(&mut col.size[0]).speed(0.05).prefix("X: "));
                                            ui.add(egui::DragValue::new(&mut col.size[1]).speed(0.05).prefix("Y: "));
                                            ui.add(egui::DragValue::new(&mut col.size[2]).speed(0.05).prefix("Z: "));
                                        });
                                    }
                                    ecs::ColliderShapeType::Sphere => {
                                        ui.add(egui::DragValue::new(&mut col.size[0]).speed(0.05).prefix("Radius: "));
                                    }
                                    ecs::ColliderShapeType::Capsule | ecs::ColliderShapeType::Cylinder => {
                                        ui.add(egui::DragValue::new(&mut col.size[0]).speed(0.05).prefix("Radius: "));
                                        ui.add(egui::DragValue::new(&mut col.size[1]).speed(0.05).prefix("Half Height: "));
                                    }
                                }

                                ui.add(egui::Slider::new(&mut col.friction, 0.0..=2.0).text("Friction"));
                                ui.add(egui::Slider::new(&mut col.restitution, 0.0..=1.0).text("Restitution (Bounciness)"));
                                ui.checkbox(&mut col.is_sensor, "Is Trigger / Sensor");
                            });
                        });
                        ui.add_space(5.0);
                    }

                    // game components
                    let dynamic_entries = world.get_component_entries();
                    for entry in &dynamic_entries {
                        if (entry.has)(world, entity_id) {
                            ui.group(|ui| {
                                ui.horizontal(|ui| {
                                    ui.label(egui::RichText::new(format!("🧩 {}", entry.name)).strong());
                                    ui.with_layout(egui::Layout::right_to_left(egui::Align::Center), |ui| {
                                        if ui.button("❌").on_hover_text(format!("Remove {}", entry.name)).clicked() {
                                            (entry.remove)(world, entity_id);
                                        }
                                    });
                                });

                                let mut visitor = EguiFieldVisitor { ui, changed: false };
                                (entry.inspect)(world, entity_id, &mut visitor);
                            });
                            ui.add_space(5.0);
                        }
                    }

                    // add component
                    ui.separator();
                    ui.menu_button("➕ Add Component", |ui| {
                        if !has_transform && ui.button("📍 Transform").clicked() {
                            world.add_transform(entity_id, Transform::default());
                            ui.close();
                        }
                        if !has_camera && ui.button("📷 Camera").clicked() {
                            world.add_camera(entity_id, CameraComponent::default());
                            ui.close();
                        }
                        if !has_light && ui.button("💡 Light Component").clicked() {
                            world.add_light(entity_id, LightComponent::default());
                            ui.close();
                        }
                        if !has_rigid_body && ui.button("📦 RigidBody").clicked() {
                            world.add_rigid_body(entity_id, RigidBodyComponent::default());
                            ui.close();
                        }
                        if !has_collider && ui.button("🛡 Collider").clicked() {
                            world.add_collider(entity_id, ColliderComponent::default());
                            ui.close();
                        }

                        let mut categories: std::collections::BTreeMap<&'static str, Vec<ecs::DynamicComponentEntry>> = std::collections::BTreeMap::new();
                        for entry in &dynamic_entries {
                            if !(entry.has)(world, entity_id) {
                                categories.entry(entry.category).or_default().push(entry.clone());
                            }
                        }

                        for (cat_name, entries_in_cat) in categories {
                            ui.separator();
                            ui.label(egui::RichText::new(format!("📁 {}", cat_name)).weak());
                            for entry in entries_in_cat {
                                if ui.button(format!("  ➕ {}", entry.name)).clicked() {
                                    (entry.add_default)(world, entity_id);
                                    ui.close();
                                }
                            }
                        }
                    });

                    ui.add_space(15.0);
                    if ui.button(egui::RichText::new("🗑 Delete Entity").color(egui::Color32::RED)).clicked() {
                        world.delete_entity(entity_id);
                        *selected_entity = None;
                    }
                });
            } else {
                ui.label(egui::RichText::new("Select an entity from Scene Hierarchy to inspect.").italics());
            }
        });

    // Performance & Camera Window
    egui::Window::new("Performance & Camera")
        .default_pos([600.0, 10.0])
        .default_size([250.0, 300.0])
        .show(ctx, |ui| {
            ui.heading("Performance");
            ui.separator();

            ui.label(format!("FPS: {:.1}", metrics.fps));
            ui.label(format!("Frame Time: {:.2} ms", metrics.frame_time_ms));
            ui.label(format!("Draw Calls: {}", metrics.draw_calls));
            ui.label(format!("Total Instances: {}", metrics.instance_count));

            ui.add_space(20.0);

            ui.heading("Camera");
            ui.separator();

            ui.label("Position");
            ui.horizontal(|ui| {
                ui.add(egui::DragValue::new(&mut camera.eye.x).speed(0.1));
                ui.add(egui::DragValue::new(&mut camera.eye.y).speed(0.1));
                ui.add(egui::DragValue::new(&mut camera.eye.z).speed(0.1));
            });

            ui.add_space(10.0);
            ui.add(egui::Slider::new(&mut camera.fovy, 10.0..=120.0).text("FOV"));
        });

    // Asset Browser Window
    egui::Window::new("Asset Browser")
        .default_pos([10.0, 600.0])
        .default_size([300.0, 200.0])
        .show(ctx, |ui| {
            let res_path = std::path::PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("../res");

            ui.heading("Models (.bin)");
            ui.separator();

            egui::ScrollArea::vertical().show(ui, |ui| {
                if let Ok(entries) = std::fs::read_dir(&res_path) {
                    for entry in entries.flatten() {
                        if let Some(name) = entry.file_name().to_str() {
                            if name.ends_with(".bin") {
                                let _ = asset::AssetId::from_path(name);

                                ui.horizontal(|ui| {
                                    ui.label(name);
                                    if ui.button("Spawn").clicked() {
                                        if let Ok(data) = std::fs::read(entry.path()) {
                                            if let Ok(baked_model) = rkyv::from_bytes::<shared::BakedModel, rkyv::rancor::Error>(&data) {
                                                let model_id = asset::AssetId::from_path(name);

                                                let spawned_id = ecs::spawn_scene(world, model_id, &baked_model, glam::Vec3::ZERO);
                                                *selected_entity = Some(spawned_id);

                                                load_requests.push(name.to_string());

                                                log::info!("Spawned {} in ECS", name);
                                            }
                                        }
                                    }
                                });
                            }
                        }
                    }
                } else {
                    ui.label(egui::RichText::new("Failed to read res/ directory").color(egui::Color32::RED));
                }
            });
        });

    if clear_requested {
        world.clear_scene();
        *selected_entity = None;

        log::info!("Scene cleared");
    }
}
